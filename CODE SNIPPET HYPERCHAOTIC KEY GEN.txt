def perturbed_logistic(x, r):
if abs(x) < 1e-15:
x += 1e-15
return abs(r * (1 - r / x)) % 1.0
def generate_CPLM(secret_key, total_points, x0, y0, z0, rX, rY, rZ):

NX = x0
NY = y0
NZ = z0
for i in range(1, len(key_bits) + 1):
kp = prefix_sum[i - 1] / 2.0
NX = (NX + kp) % 1
NY = (NY + kp) % 1
NZ = (NZ + kp) % 1
X = [NX, perturbed_logistic(NX, rX), perturbed_logistic(perturbed_logistic(NX, rX), rX)]
Y = [NY, perturbed_logistic(NY, rY), perturbed_logistic(perturbed_logistic(NY, rY), rY)]
Z = [NZ, perturbed_logistic(NZ, rZ), perturbed_logistic(perturbed_logistic(NZ, rZ), rZ)]

break

def inverse_permutation_matrix(PM):
COL, ROW, _ = PM.shape
IPM = np.zeros_like(PM)
for i in range(1, COL + 1):
for j in range(1, ROW + 1):
pi = PM[i - 1, j - 1, 0]
pj = PM[i - 1, j - 1, 1]
IPM[pi - 1, pj - 1, 0] = i
IPM[pi - 1, pj - 1, 1] = j
return IPM
def flat_index(i, j, ROW):

def paper_sum_enc(PD_flat, C_flat, k):
s = 0
start = max(0, k - 8)
for n in range(start, k + 1):
s += int(PD_flat[n]) + int(C_flat[n])
return s % 256
def paper_sum_dec(PD_flat, C_flat, k):

s += int(PD_flat[n]) + int(C_flat[n])
return s

prob = prob[prob > 0]

def adjacent_correlation(image, direction='horizontal'):

if direction == 'horizontal':
x = img[:, :-1].flatten()
y = img[:, 1:].flatten()
elif direction == 'vertical':

x = img[:-1, :-1].flatten()

mean_x = np.mean(x)
mean_y = np.mean(y)
num = np.sum((x - mean_x) * (y - mean_y))
den = np.sqrt(np.sum((x - mean_x)*2) * np.sum((y - mean_y)*2)) + 1e-12
return num / den
